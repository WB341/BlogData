class BaseClass{
    public:
    void greet(){
        cout<<"这是基类"<<endl;
    }
};
class SubClass: public BaseClass
{
    public:
    void greet()
    {
        cout<<"这是子类"<<endl;
    }
}

int main(void){
    SubClass obj;
    BaseClass*pBase=&obj;
    BaseClass&refBase=obj;

    obj.greet();
    pBase->greet();
    refBase.greet();
}
